cmake_minimum_required(VERSION 3.16)
project(ego_nav2_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep nav2_smac_planner nav2_behaviors nav2_msgs nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib tf2_geometry_msgs tf2_ros nav_msgs geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
set(P vendor/planner)
set(CORE_INCLUDES include ${P}/include ${P}/GridMap2D/include ${P}/GridMap2D/include/plan_env ${P}/path_searching/include ${P}/bspline_opt/include ${P}/plan_manage/include ${P}/plan_manage/include/plan_manage)
add_library(ego_core STATIC ${P}/GridMap2D/src/grid_map.cpp ${P}/path_searching/src/dyn_a_star.cpp ${P}/bspline_opt/src/uniform_bspline.cpp ${P}/bspline_opt/src/bspline_optimizer.cpp ${P}/bspline_opt/src/gradient_descent_optimizer.cpp ${P}/plan_manage/src/planner_interface.cpp)
target_include_directories(ego_core PUBLIC ${CORE_INCLUDES})
target_link_libraries(ego_core Eigen3::Eigen)
add_library(ego_controller SHARED src/controller.cpp src/behaviors.cpp src/obstacle_memory.cpp src/exact_goal_planner.cpp)
target_include_directories(ego_controller PRIVATE ${CORE_INCLUDES})
target_link_libraries(ego_controller ego_core)
ament_target_dependencies(ego_controller nav2_smac_planner nav2_behaviors nav2_msgs nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib tf2_geometry_msgs tf2_ros nav_msgs geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d costmap_plugins.xml)
install(TARGETS ego_controller LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md vendor/UPSTREAM.md vendor/LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)

  ament_add_gtest(test_ego_core test/test_core.cpp)
  target_link_libraries(test_ego_core ego_core)
  ament_add_gtest(test_control test/test_control.cpp)
  target_include_directories(test_control PRIVATE include)
  target_compile_definitions(test_control PRIVATE EGO_LATTICE_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../mid360_nav2/config/smac_lattice_diff_5cm_0p5m.json")
  ament_target_dependencies(test_control nav2_core nav2_costmap_2d nav2_util pluginlib rclcpp rclcpp_lifecycle tf2_ros)
  ament_add_gtest(test_turn_profile test/test_turn_profile.cpp)
  target_include_directories(test_turn_profile PRIVATE include)
  ament_add_gtest(test_precise_approach test/test_precise_approach.cpp)
  target_include_directories(test_precise_approach PRIVATE include)
  ament_add_gtest(test_smooth_detour test/test_smooth_detour.cpp)
  target_include_directories(test_smooth_detour PRIVATE include)
  ament_add_gtest(test_speed_envelope test/test_speed_envelope.cpp)
  target_include_directories(test_speed_envelope PRIVATE include)
  add_executable(test_spin_bag_replay test/test_spin_bag_replay.cpp)
  target_include_directories(test_spin_bag_replay PRIVATE include)

ament_package()
